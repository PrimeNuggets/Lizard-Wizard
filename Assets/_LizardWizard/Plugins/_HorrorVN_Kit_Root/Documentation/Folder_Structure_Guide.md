Assets/
└── HorrorVN_Kit/             # Root folder to keep custom assets separate from Unity defaults.
    ├── Audio/                # SFX and background music.
    ├── Fonts/                # Custom TextMeshPro fonts.
    ├── Materials/            # Shaders and materials.
    ├── Prefabs/              # Drag-and-drop objects.
    │   ├── Interactables/    # 3D objects to click on (diaries, doors).
    │   └── UI/               # Main Canvas, dialogue boxes, keypads.
    ├── Scenes/               # Unity scene files.
    ├── Scripts/              # C# scripts.
    │   ├── Core/             # Main system scripts.
    │   └── Triggers/         # Scripts attached to objects in the scene.
    └── Sprites/              # 2D image assets.
        ├── Backgrounds/      # Room environments.
        ├── Characters/       # Character portraits.
        └── UI_Elements/      # Text boxes, buttons, cursors.
